package h2transport

import (
	"net/http"
	"testing"
)

func BenchmarkShutdownNoConnections(b *testing.B) {
	b.Run("fresh", func(b *testing.B) {
		handler := http.NotFoundHandler()
		b.ReportAllocs()
		for b.Loop() {
			server, err := NewServer(handler)
			if err != nil {
				b.Fatal(err)
			}
			if err := server.Shutdown(b.Context()); err != nil {
				b.Fatal(err)
			}
		}
	})
	b.Run("repeated", func(b *testing.B) {
		server, err := NewServer(http.NotFoundHandler())
		if err != nil {
			b.Fatal(err)
		}
		if err := server.Shutdown(b.Context()); err != nil {
			b.Fatal(err)
		}
		b.ReportAllocs()
		for b.Loop() {
			if err := server.Shutdown(b.Context()); err != nil {
				b.Fatal(err)
			}
		}
	})
}

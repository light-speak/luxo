import Foundation
@testable import LuxoClient

@main
struct GoConsumer {
  static let selection = "value, duration, optional, values, label"

  static func main() async {
    do {
      try await run()
    } catch {
      FileHandle.standardError.write(Data("Go consumer failed: \(error)\n".utf8))
      exit(1)
    }
  }

  static func run() async throws {
    try verifyNestedBoundaries()
    try await verifyNullableRootBoundaries()
    let environment = ProcessInfo.processInfo.environment
    guard let endpoint = environment["LUXO_INTEROP_ENDPOINT"],
      let corpus = environment["LUXO_INTEROP_CORPUS"]
    else {
      throw failure("missing live Go fixture environment")
    }
    let gateway: Bool
    switch environment["LUXO_INTEROP_TOPOLOGY"] {
    case "direct": gateway = false
    case "gateway": gateway = true
    default: throw failure("missing or invalid topology; use the Go -run owner")
    }
    let scope =
      gateway
      ? "gateway success/streams/errors" : "direct success/streams/errors"
    let cases = try JSONDecoder().decode(
      Corpus.self, from: Data(contentsOf: URL(fileURLWithPath: corpus)))
    let values = try cases.int64Cases.map { item in
      guard let value = Int(item.value) else {
        throw failure("integer outside platform range: \(item.value)")
      }
      return value
    }
    for websocket in [false, true] {
      for binary in [false, true] {
        try await verifyChannel(
          endpoint: endpoint, websocket: websocket, binary: binary, values: values,
          numberTokens: cases.jsonNumberTokens, dates: cases.dateTimeCases)
        print("\(scope): \(websocket ? "WebSocket" : "HTTP") \(binary ? "Binary" : "JSON") passed")
      }
    }
    for binary in [false, true] {
      try await verifyTokenRefresh(endpoint: endpoint, binary: binary)
      print("\(scope): HTTP \(binary ? "Binary" : "JSON") access/refresh lifecycle passed")
      for start in [UInt64(0), 9_007_199_254_740_990, UInt64(Int64.max) - 1, UInt64.max - 2] {
        try await verifySubscriptionIDs(endpoint: endpoint, binary: binary, start: start)
        print("\(scope): concurrent \(binary ? "Binary" : "JSON") UInt64 subscriptions from \(start) passed")
      }
      for fail in [false, true] {
        let socket = try WebSocketTransport(
          url: endpoint.replacingOccurrences(of: "http:", with: "ws:"), timeout: 5,
          session: URLSession(configuration: .ephemeral), initialSequence: .max - 1)
        defer { socket.close() }
        let client = try LuxoGeneratedClient(transport: socket)
        if binary { client.enableBinary() }
        socket.connect()
        try await verifyFiniteStream(client, fail: fail, reuse: false)
        print("\(scope): \(binary ? "Binary" : "JSON") UInt64.max terminal failure=\(fail) passed")
      }
    }
  }

  static func verifyTokenRefresh(endpoint: String, binary: Bool) async throws {
    let session = URLSession(configuration: .ephemeral)
    defer { session.invalidateAndCancel() }
    let bootstrapTransport = try URLSessionTransport(endpoint: endpoint, timeout: 5, session: session)
    let transport = try URLSessionTransport(endpoint: endpoint, timeout: 5, session: session)
    let bootstrap = try LuxoGeneratedClient(transport: bootstrapTransport)
    let client = try LuxoGeneratedClient(transport: transport)
    if binary { bootstrap.enableBinary(); client.enableBinary() }
    let refresh = try await bootstrap.issueFixtureRefresh()
    let access = try await bootstrap.issueFixtureAccess()
    var rejectedAccess = false
    do {
      _ = try await bootstrap.exchangeFixtureRefresh(token: access)
    } catch is LuxoError { rejectedAccess = true }
    try ensure(rejectedAccess, "access token accepted by refresh endpoint")
    transport.setToken(refresh)
    try await expectUnauthorized(client)
    let attempts = TerminalCounter()
    transport.onTokenExpired = {
      attempts.increment()
      return try? await bootstrap.exchangeFixtureRefresh(token: refresh)
    }
    for _ in 0..<2 {
      try ensure(try await client.fixtureIdentity() == 9_007_199_254_740_993, "refreshed identity lost precision")
    }
    try ensure(attempts.value == 1, "successful refresh was repeated")
    transport.setToken(refresh)
    transport.onTokenExpired = { attempts.increment(); return nil }
    try await expectUnauthorized(client)
    try ensure(attempts.value == 2, "declined refresh looped or was skipped")
    transport.onTokenExpired = { attempts.increment(); return refresh }
    try await expectUnauthorized(client)
    try ensure(attempts.value == 3, "invalid renewed token retried indefinitely")
  }

  static func expectUnauthorized(_ client: LuxoGeneratedClient) async throws {
    do {
      _ = try await client.fixtureIdentity()
      throw failure("refresh token accepted as API access")
    } catch let error as LuxoError {
      try ensure(error.code == 401, "unauthorized error changed: \(error)")
    }
  }

  static func verifySubscriptionIDs(endpoint: String, binary: Bool, start: UInt64) async throws {
    let socket = try WebSocketTransport(
      url: endpoint.replacingOccurrences(of: "http:", with: "ws:"), timeout: 5,
      session: URLSession(configuration: .ephemeral), initialSequence: start)
    defer { socket.close() }
    let client = try LuxoGeneratedClient(transport: socket)
    if binary { client.enableBinary() }
    socket.connect()
    let (first, firstContinuation) = AsyncStream<Precision>.makeStream()
    let (second, secondContinuation) = AsyncStream<Precision>.makeStream()
    defer { firstContinuation.finish(); secondContinuation.finish() }
    async let firstSetup = client.subscribeWatchPrecision(value: Int.min, select: selection) {
      firstContinuation.yield($0)
    }
    async let secondSetup = client.subscribeWatchPrecision(value: Int.max, select: selection) {
      secondContinuation.yield($0)
    }
    let cancellations = try await (firstSetup, secondSetup)
    defer { cancellations.0(); cancellations.1() }
    try ensure(try await firstValue(first).value.requireValue() == Int.min, "first subscription identity changed")
    try ensure(try await firstValue(second).value.requireValue() == Int.max, "second subscription identity changed")
    if start == UInt64.max - 2 {
      do {
        _ = try await client.subscribeWatchPrecision(value: 1, select: selection) { _ in }
        throw failure("UInt64 subscription counter wrapped")
      } catch let error as LuxoError {
        try ensure(error.name == "RequestIDExhausted", "wrong exhaustion error")
      }
    }
  }

  static func verifyChannel(
    endpoint: String, websocket: Bool, binary: Bool, values: [Int], numberTokens: [String],
    dates: [DateTimeCase]
  ) async throws {
    let socket =
      try websocket
      ? WebSocketTransport(url: endpoint.replacingOccurrences(of: "http:", with: "ws:"), timeout: 5)
      : nil
    let transport: any Transport
    if let socket {
      transport = socket
    } else {
      transport = try URLSessionTransport(endpoint: endpoint, timeout: 5)
    }
    defer { socket?.close() }
    let observed = RecordingTransport(base: transport)
    let client = try LuxoGeneratedClient(transport: observed)
    try ensure(observed.registrationCount == 1, "schema was not registered at initialization")
    if binary { client.enableBinary() }
    try ensure(observed.registrationCount == 1, "mode change repeated schema registration")
    socket?.connect()
    if ProcessInfo.processInfo.environment["LUXO_INTEROP_ENUM_UPGRADE"] == "true" {
      try await verifyEnumUpgrade(client, websocket: websocket)
      return
    }
    try await verifyUnary(client, values: values)
    try await verifyNullableRoots(client, values: values)
    try await verifyTimestamps(client, cases: dates)
    try await verifyBusinessFloats(client)
    try await verifyJSONNumbers(client, tokens: numberTokens, websocket: websocket)
    try await verifyNamedEnums(client, websocket: websocket)
    try ensure(
      try await client.checkListEffects() == [1, 2, 3, 3], "list evaluation count mismatch")
    try await verifyErrors(client, values: values)
    try await verifyStrictQueries(client)
    try await verifyJSONErrors(client, tokens: numberTokens)
    try await verifyPagination(client)
    try await verifyNullableArrays(client, websocket: websocket)
    try await verifyNestedResponses(client, websocket: websocket)
    try await verifyScalarArrays(client, websocket: websocket)
    if websocket {
      try await verifyNullableRootStreams(client)
      try await verifyTimestampStreams(client, cases: dates)
      try await verifyStreams(client, values: values)
      for fail in [false, true] { try await verifyFiniteStream(client, fail: fail) }
    }
  }

  static let scalarSelection =
    "ints, floats, strings, bools, dates, durations, uuids, bytes, decimals, jsons, enums, tail"

  static func verifyNullableRoots(_ client: LuxoGeneratedClient, values: [Int]) async throws {
    for value: Int? in [nil, 0] + values.map(Optional.some) {
      try ensure(try await client.echoOptionalInteger(value: value) == value, "nullable integer mismatch")
    }
    for values: [Int]? in [nil, [], values] {
      try ensure(try await client.echoOptionalIntegers(values: values) == values, "nullable list mismatch")
    }
    let input = makeInput(value: 42, optional: nil, values: [])
    let missing = try await client.echoOptionalPrecision(input: nil, select: "value")
    try ensure(missing == nil, "nullable model became a value")
    let row = try await client.echoOptionalPrecision(input: input, select: "value")
    try ensure(try row?.value.requireValue() == 42, "nullable model field mismatch")
    guard case .unselected? = row?.label else { throw failure("unselected field was returned") }
    let absent = try await client.echoOptionalPrecisions(inputs: nil, select: "value")
    try ensure(absent == nil, "nullable model list became a value")
    for inputs in [[], [input]] {
      let rows = try await client.echoOptionalPrecisions(inputs: inputs, select: "value")
      try ensure(rows?.count == inputs.count, "nullable model list size mismatch")
      for row in rows ?? [] { try ensure(try row.value.requireValue() == 42, "nullable list field mismatch") }
    }
  }

  static func verifyNullableRootStreams(_ client: LuxoGeneratedClient) async throws {
    let integers: [Int?] = try await nullableStream { observer, receive in
      try await client.subscribeWatchOptionalInteger(observer: observer, handler: receive)
    }
    try ensure(integers == [nil, 0], "null ended integer stream")
    let lists: [[Int]?] = try await nullableStream { observer, receive in
      try await client.subscribeWatchOptionalIntegers(observer: observer, handler: receive)
    }
    try ensure(lists == [nil, []], "null ended list stream")
    let records: [Precision?] = try await nullableStream { observer, receive in
      try await client.subscribeWatchOptionalPrecision(select: "value", observer: observer, handler: receive)
    }
    try ensure(records.count == 2 && records[0] == nil, "null ended model stream")
    try ensure(try records[1]?.value.requireValue() == 42, "nullable stream field mismatch")
    guard case .unselected? = records[1]?.label else { throw failure("nullable stream selection ignored") }
    try ensure(try await client.echoInteger(value: 42) == 42, "completion closed peer connection")
  }

  static func nullableStream<Value: Sendable>(
    _ subscribe: (StreamObserver, @escaping @Sendable (Value?) -> Void) async throws -> (@Sendable () -> Void)
  ) async throws -> [Value?] {
    let (events, continuation) = AsyncThrowingStream<Value?, Error>.makeStream()
    let terminals = TerminalCounter()
    let observer = StreamObserver(onComplete: { terminals.increment(); continuation.finish() }, onError: {
      terminals.increment(); continuation.finish(throwing: $0)
    })
    let cancel = try await subscribe(observer) { continuation.yield($0) }
    defer { cancel(); continuation.finish() }
    let rows = try await withThrowingTaskGroup(of: [Value?].self) { group in
      group.addTask {
        var values: [Value?] = []
        for try await value in events { values.append(value) }
        return values
      }
      group.addTask { try await Task.sleep(nanoseconds: 5_000_000_000); throw failure("nullable stream timeout") }
      defer { group.cancelAll() }
      guard let values = try await group.next() else { throw failure("missing nullable stream") }
      return values
    }
    try ensure(terminals.value == 1, "nullable stream terminal count mismatch")
    return rows
  }

  static func verifyNullableRootBoundaries() async throws {
    try await verifyNullableStructuredBoundaries()
    for data in [Data(), Data([1]), Data([2]), Data([0, 0])] {
      let client = try LuxoGeneratedClient(transport: NullablePayloadTransport(data: data, mode: .binary))
      for kind in 0..<4 {
        do {
          switch kind {
          case 0: _ = try await client.echoOptionalInteger(value: nil)
          case 1: _ = try await client.echoOptionalIntegers(values: nil)
          case 2: _ = try await client.echoOptionalPrecision(input: nil, select: "value")
          default: _ = try await client.echoOptionalPrecisions(inputs: nil, select: "value")
          }
          throw failure("invalid nullable envelope accepted")
        } catch let error as LuxoError { try ensure(error.name == "DecodeError", "wrong nullable failure") }
      }
    }
    for data in [Data([1, 84, 0]), Data([1, 0x80])] {
      let client = try LuxoGeneratedClient(transport: NullablePayloadTransport(data: data, mode: .binary))
      do { _ = try await client.echoOptionalInteger(value: nil); throw failure("invalid nullable value accepted") }
      catch let error as LuxoError { try ensure(error.name == "DecodeError", "wrong nullable payload failure") }
    }
  }

  static func verifyNullableStructuredBoundaries() async throws {
    guard let field = LUXO_SCHEMA["echoOptionalPrecision"]?.fields["value"]?.fieldID else {
      throw failure("missing nullable response field metadata")
    }
    var body = Encoder()
    body.writeVarint(UInt64(field)); body.writeSvarint(42); body.writeVarint(0)
    for list in [false, true] {
      let valid = Data(list ? [1, 1, 0] : [1, 0]) + body.data
      let client = try LuxoGeneratedClient(transport: NullablePayloadTransport(data: valid, mode: .binary))
      let value = try await nullableStructuredValue(client, list: list)
      try ensure(value == 42, "nullable structured root mismatch")
      for data in [Data(valid.dropLast()), valid + Data([99])] {
        let invalid = try LuxoGeneratedClient(transport: NullablePayloadTransport(data: data, mode: .binary))
        do { _ = try await nullableStructuredValue(invalid, list: list); throw failure("bad nullable structure accepted") }
        catch let error as LuxoError { try ensure(error.name == "DecodeError", "wrong nullable structure error") }
      }
      for mode in [TransportMode.json, .binary] {
        let data = mode == .json ? Data("null".utf8) : Data([0])
        let absent = try LuxoGeneratedClient(transport: NullablePayloadTransport(data: data, mode: mode))
        try ensure(try await nullableStructuredValue(absent, list: list) == nil, "null structure became a value")
      }
    }
  }

  static func nullableStructuredValue(_ client: LuxoGeneratedClient, list: Bool) async throws -> Int? {
    if list { return try await client.echoOptionalPrecisions(inputs: nil, select: "value")?.first?.value.requireValue() }
    return try await client.echoOptionalPrecision(input: nil, select: "value")?.value.requireValue()
  }

  static func verifyTimestamps(_ client: LuxoGeneratedClient, cases: [DateTimeCase]) async throws {
    for item in cases {
      try ensure(try await client.echoTimestamp(value: item.input) == item.canonical, "DateTime mismatch: \(item.input)")
    }
    let inputs = cases.map(\.input)
    let expected = cases.map(\.canonical)
    try ensure(try await client.echoTimestamps(values: inputs) == expected, "DateTime list mismatch")
    let input = ScalarArraysInput(dates: inputs, tail: 42)
    let row = try await client.echoScalarArrays(input: input, select: "dates, tail")
    try ensure(try row.dates.requireValue() == expected, "DateTime input mismatch")
    for row in try await client.listScalarArrays(input: input, select: "dates, tail") {
      try ensure(try row.dates.requireValue() == expected, "DateTime column mismatch")
      try ensure(try row.tail.requireValue() == 42, "DateTime column cursor mismatch")
    }
  }

  static func verifyTimestampStreams(_ client: LuxoGeneratedClient, cases: [DateTimeCase]) async throws {
    for item in cases {
      let (stream, continuation) = AsyncStream<String>.makeStream()
      let cancel = try await client.subscribeWatchTimestamp(value: item.input) { continuation.yield($0) }
      defer { cancel(); continuation.finish() }
      try ensure(try await firstValue(stream) == item.canonical, "DateTime scalar stream changed nanoseconds")
    }
    for items in [[], Array(cases.prefix(1)), cases] {
      let (stream, continuation) = AsyncStream<[String]>.makeStream()
      let cancel = try await client.subscribeWatchTimestamps(values: items.map(\.input)) { continuation.yield($0) }
      defer { cancel(); continuation.finish() }
      try ensure(try await firstValue(stream) == items.map(\.canonical), "DateTime list stream changed nanoseconds")
    }
    try ensure(try await client.echoInteger(value: 42) == 42, "DateTime stream cancellation closed the connection")
  }

  static func verifyBusinessFloats(_ client: LuxoGeneratedClient) async throws {
    let values: [Double] = [0, -0.0, .leastNonzeroMagnitude, -.leastNonzeroMagnitude, .greatestFiniteMagnitude, -.greatestFiniteMagnitude]
    for value in values {
      let result = try await client.scaleFloat(value: value, factor: 1)
      try ensure(result.bitPattern == value.bitPattern, "business Float bits changed")
      let rows = try await client.scaleFloatList(value: value, factor: 1)
      try ensure(rows.count == 2 && rows.allSatisfy { $0.bitPattern == value.bitPattern }, "Float list bits changed")
    }
    for value in [Double.greatestFiniteMagnitude, -Double.greatestFiniteMagnitude] {
      for list in [false, true] {
        do {
          if list { _ = try await client.scaleFloatList(value: value, factor: 2) }
          else { _ = try await client.scaleFloat(value: value, factor: 2) }
          throw failure("business Float overflow was accepted")
        } catch let error as LuxoError {
          try ensure(error.code == 500 && error.name == "Internal" && error.message == "error.internal", "overflow error identity changed")
          try ensure(error.data == nil && error.cause == nil, "overflow exposed internal data")
        }
        try ensure(try await client.scaleFloat(value: 1.25, factor: 2) == 2.5, "success after overflow failed")
      }
    }
  }

  static func jsonFixtureDocuments(_ token: String) -> [Data] {
    [Data(token.utf8), Data("{\"nested\":[\(token),null,{\"n\":\(token)}]}".utf8)]
  }

  static func verifyJSONNumbers(
    _ client: LuxoGeneratedClient, tokens: [String], websocket: Bool
  ) async throws {
    for token in tokens {
      for data in jsonFixtureDocuments(token) {
        let value = try JSONValue.parse(data)
        let result = try await client.echoJSON(value: value)
        try ensure(try result.jsonData() == data, "generic JSON number changed: \(token)")
        let list = try await client.echoJSONs(values: [value, .null])
        try ensure(list == [value, .null], "generic JSON list changed: \(token)")
        if websocket { try await verifyJSONStreams(client, value: value) }
      }
    }
  }

  static func verifyJSONErrors(_ client: LuxoGeneratedClient, tokens: [String]) async throws {
    for token in tokens {
      for data in jsonFixtureDocuments(token) {
        let value = try JSONValue.parse(data)
        try await verifyJSONError(client, value: value, token: token)
      }
    }
  }

  static func verifyJSONError(
    _ client: LuxoGeneratedClient, value: JSONValue, token: String
  ) async throws {
    do {
      _ = try await client.rejectJSON(value: value)
      throw failure("expected JSONRejected")
    } catch let error as LuxoError {
      try ensure(error.name == "JSONRejected" && error.code == 422, "JSON error identity mismatch")
      try ensure(error.message == "error.json_rejected", "JSON error message mismatch")
      try ensure(
        error.traceId?.isEmpty == false && error.cause == nil, "JSON error metadata mismatch")
      guard let data = error.data as? [String: Any] else { throw failure("missing JSON error data") }
      if case .object = value {
        guard let opaque = data["value"] as? [String: Any],
          let items = opaque["nested"] as? [Any], items.count == 3,
          let nested = items[2] as? [String: Any]
        else { throw failure("missing nested JSON error data") }
        try ensure(items[1] is NSNull, "JSON error null changed")
        try ensure(rawNumberToken(items[0]) == token, "JSON error number changed: \(token)")
        try ensure(rawNumberToken(nested["n"]) == token, "nested JSON error number changed: \(token)")
      } else {
        try ensure(rawNumberToken(data["value"]) == token, "scalar JSON error number changed: \(token)")
      }
    }
    try ensure(try await client.echoJSON(value: value) == value, "success after JSON error changed")
  }

  static func rawNumberToken(_ value: Any?) -> String? {
    if let number = value as? JSONNumber { return number.literal }
    if let integer = value as? Int64 { return String(integer) }
    return nil
  }

  static func verifyJSONStreams(_ client: LuxoGeneratedClient, value: JSONValue) async throws {
    let (stream, continuation) = AsyncStream<JSONValue>.makeStream()
    let cancel = try await client.subscribeWatchJSON(value: value) { continuation.yield($0) }
    defer {
      cancel()
      continuation.finish()
    }
    try ensure(try await firstValue(stream) == value, "generic JSON stream number changed")
    let (rows, listContinuation) = AsyncStream<[JSONValue]>.makeStream()
    let cancelList = try await client.subscribeWatchJSONs(values: [value, .null]) {
      listContinuation.yield($0)
    }
    defer {
      cancelList()
      listContinuation.finish()
    }
    try ensure(try await firstValue(rows) == [value, .null], "generic JSON list stream changed")
  }

  static func verifyNamedEnums(_ client: LuxoGeneratedClient, websocket: Bool) async throws {
    for value in FixtureState.knownCases {
      try ensure(try await client.echoFixtureState(value: value) == value, "enum value mismatch")
    }
    for values: [FixtureState] in [[], [.open], [.closed, .open]] {
      try ensure(try await client.echoFixtureStates(values: values) == values, "enum list mismatch")
    }
    for raw in ["FUTURE", "", "open"] {
      let value = FixtureState(rawValue: raw)
      for list in [false, true] {
        do {
          if list {
            _ = try await client.echoFixtureStates(values: [.open, value])
          } else {
            _ = try await client.echoFixtureState(value: value)
          }
          throw failure("server accepted an undeclared enum member")
        } catch let error as LuxoError {
          try ensure(error.code == 400, "undeclared enum member did not return 400")
        }
      }
      try ensure(try await client.echoFixtureState(value: .open) == .open, "success after enum rejection")
    }
    guard websocket else { return }
    let (stream, continuation) = AsyncStream<FixtureState>.makeStream()
    let cancel = try await client.subscribeWatchFixtureState { continuation.yield($0) }
    defer {
      cancel()
      continuation.finish()
    }
    try ensure(try await firstValue(stream) == .open, "enum stream mismatch")
    let (listStream, listContinuation) = AsyncStream<[FixtureState]>.makeStream()
    let cancelList = try await client.subscribeWatchFixtureStates { listContinuation.yield($0) }
    defer {
      cancelList()
      listContinuation.finish()
    }
    try ensure(try await firstValue(listStream) == [.open, .closed], "enum list stream mismatch")
  }

  static func verifyFutureState(_ value: FixtureState) throws {
    guard case .unknown(let raw) = value, raw == "FUTURE" else {
      throw failure("future enum was not explicitly Unknown")
    }
  }

  static func verifyEnumUpgrade(_ client: LuxoGeneratedClient, websocket: Bool) async throws {
    try await verifyPagination(client)
    let future = FixtureState(rawValue: "FUTURE")
    try verifyFutureState(future)
    let received = try await client.echoFixtureState(value: future)
    try verifyFutureState(received)
    try verifyFutureState(try await client.echoFixtureState(value: received))
    let expected: [FixtureState] = [.open, future, .closed]
    let values = try await client.echoFixtureStates(values: expected)
    try ensure(values == expected, "future enum mixed list changed")
    try verifyFutureState(values[1])
    try ensure(try await client.echoFixtureStates(values: []).isEmpty, "empty enum list changed")
    if websocket { try await verifyFutureStreams(client, expected: expected) }
    do {
      _ = try await client.echoFixtureState(value: FixtureState(rawValue: "UNDECLARED"))
      throw failure("server accepted an undeclared enum")
    } catch let error as LuxoError {
      try ensure(error.code == 400 && error.name == "BadRequest", "wrong enum error")
    }
    try ensure(try await client.echoInteger(value: 42) == 42, "request after enum failure changed")
  }

  static func verifyFutureStreams(_ client: LuxoGeneratedClient, expected: [FixtureState]) async throws {
    let (stream, continuation) = AsyncStream<FixtureState>.makeStream()
    let cancel = try await client.subscribeWatchFixtureState { continuation.yield($0) }
    defer { cancel(); continuation.finish() }
    try verifyFutureState(try await firstValue(stream))
    let (list, listContinuation) = AsyncStream<[FixtureState]>.makeStream()
    let cancelList = try await client.subscribeWatchFixtureStates { listContinuation.yield($0) }
    defer { cancelList(); listContinuation.finish() }
    let rows = try await firstValue(list)
    try ensure(rows == expected, "future enum stream list changed")
    try verifyFutureState(rows[1])
  }

  static func verifyErrors(_ client: LuxoGeneratedClient, values: [Int]) async throws {
    for value in values {
      do {
        _ = try await client.rejectPrecision(value: value)
        throw failure("expected PrecisionRejected")
      } catch let error as LuxoError {
        try ensure(
          error.name == "PrecisionRejected" && error.code == 422, "error identity mismatch")
        try ensure(error.message == "error.precision_rejected", "error message mismatch")
        try ensure(error.traceId?.isEmpty == false, "missing error trace ID")
        guard let data = error.data as? [String: Any], let integer = data["value"] as? NSNumber
        else {
          throw failure("missing numeric error data")
        }
        try ensure(integer.stringValue == String(value), "error integer lost precision")
        try ensure(data["label"] as? String == "精度 café", "error label mismatch")
        try ensure(error.cause == nil, "unexpected error cause")
      }
      try ensure(
        try await client.echoInteger(value: value) == value, "success after error mismatch")
    }
  }

  static func verifyStrictQueries(_ client: LuxoGeneratedClient) async throws {
    guard case .int64(9007199254740993) = LuxoQuery_listQueryProbes.count_eq(9007199254740993).value else {
      throw failure("query Int64 lost precision")
    }
    try ensure(LuxoQuery_listQueryProbes.count_desc().order == "desc", "query sorter changed")
    try await verifyQueryError(client, field: "createdAt") {

  static func verifyConcurrentSelections(_ client: LuxoGeneratedClient) async throws {
    let valuePlan = try LuxoGeneratedClient.prepareSelection_echoPrecision("value")
    let labelPlan = try LuxoGeneratedClient.prepareSelection_echoPrecision("label")
    let combinedPlan = try LuxoGeneratedClient.prepareSelection_echoPrecision("value, label")
    try await withThrowingTaskGroup(of: Void.self) { group in
      for index in 0..<16 {
        group.addTask {
          let input = PrecisionInput(
            value: Int.max - index, duration: Int64(index), optional: nil, values: [],
            label: "call-\(index)", ignored: 0)

            input: input, select: prepared ? nil : "label", selectionPlan: prepared ? labelPlan : nil)
          async let combined = client.echoPrecision(
            input: input, select: "value, label", selectionPlan: prepared ? combinedPlan : valuePlan)
          let rows = try await [value, label, combined]
          try ensure(try rows[0].value.requireValue() == input.value, "concurrent value mismatch")
          try ensure(unselected(rows[0].label), "concurrent value gained label")
          try ensure(try rows[1].label.requireValue() == input.label, "concurrent label mismatch")
          try ensure(unselected(rows[1].value), "concurrent label gained value")

            try rows[2].value.requireValue() == input.value, "concurrent combined value mismatch")

        }
      }
      try await group.waitForAll()

      input: makeInput(value: Int.max, optional: nil, values: []), selectionPlan: valuePlan)

  static func firstValue<Value: Sendable>(_ stream: AsyncStream<Value>) async throws -> Value {
    try await withThrowingTaskGroup(of: Value.self) { group in
      group.addTask {
        for await row in stream { return row }
        throw failure("stream ended without data")
      }
      group.addTask {
        try await Task.sleep(nanoseconds: 5_000_000_000)
        throw failure("stream timed out")
      }
      defer { group.cancelAll() }
      guard let row = try await group.next() else { throw failure("missing stream result") }
      return row
    }
  }

  static func verifyFiniteStream(_ client: LuxoGeneratedClient, fail: Bool, reuse: Bool = true) async throws {
    let text = #"{"n":123456789012345678901234567890,"fraction":1.2345678901234567890123456789,"zero":-0}"#
    let value = try JSONValue.parse(Data(text.utf8))
    let (events, continuation) = AsyncStream<FiniteEvent>.makeStream(bufferingPolicy: .bufferingOldest(65))
    let count = TerminalCounter()
    let observer = StreamObserver(onComplete: {
      count.increment(); continuation.yield(.terminal(nil)); continuation.finish()
    }, onError: { error in
      count.increment(); continuation.yield(.terminal(error)); continuation.finish()
    })
    let cancel = try await client.subscribeWatchFiniteJSON(value: value, fail: fail, observer: observer) {
      continuation.yield(.row($0))
    }
    defer { cancel(); cancel(); continuation.finish() }
    try await withThrowingTaskGroup(of: Void.self) { group in
      group.addTask {
        var rows = 0
        for await event in events {
          switch event {
          case .row(let row):
            try ensure(row.count == 2 && row[0] == .integer(Int64(rows)) && row[1] == value, "finite row changed")
            rows += 1
          case .terminal(let error):
            try ensure(rows == 64, "terminal overtook accepted data: \(rows)")
            if fail {
              try ensure(error?.name == "JSONRejected" && error?.code == 422, "finite error identity changed")
              guard let data = error?.data as? [String: Any], let opaque = data["value"] as? [String: Any]
              else { throw failure("missing finite error data") }
              try ensure(rawNumberToken(opaque["n"]) == "123456789012345678901234567890", "finite integer changed")
              try ensure(rawNumberToken(opaque["fraction"]) == "1.2345678901234567890123456789", "finite fraction changed")
              try ensure(rawNumberToken(opaque["zero"]) == "-0", "finite negative zero changed")
            } else { try ensure(error == nil, "normal stream failed") }
            return
          }
        }
        throw failure("finite stream closed without terminal")
      }
      group.addTask { try await Task.sleep(nanoseconds: 5_000_000_000); throw failure("finite stream timed out") }
      defer { group.cancelAll() }
      _ = try await group.next()
    }
    if reuse { try ensure(try await client.echoJSON(value: value) == value, "connection failed after terminal") }
    try ensure(count.value == 1, "terminal notified more than once")
  }

  // Malformed payload probes use the same response boundary as generated API methods.
  static func verifyNestedBoundaries() throws {
    for columnar in [false, true] {
      func read(_ bytes: [UInt8]) throws -> NestedGroup {
        if columnar { return try decodeColumnarNestedGroup(Data(bytes))[0] }
        return try decodeLuxoResponse(Data(bytes)) { try decodeNestedGroup(&$0) }
      }
      for field: UInt8 in 2...5 {
        let child: [UInt8] = field < 4 ? [0, 1, 84, 0] : [1, 0, 1, 84, 0]
        func wrap(_ payload: [UInt8]) -> [UInt8] {
          var bytes: [UInt8] = columnar ? [1, 0] : [0]
          bytes.append(field)
          if field == 3 || field == 5 { bytes.append(1) }
          bytes.append(UInt8(payload.count))
          return bytes + payload + [6, 84, 0]
        }
        let valid = wrap(child)
        let row = try read(valid)
        try ensure(try row.tail.requireValue() == 42, "boundary tail mismatch")
        let value: Precision?
        switch field {
        case 2: value = try row.primary.requireValue()
        case 3: value = try row.optional.requireValue()
        case 4: value = try row.rows.requireValue().first
        default: value = try row.optionalRows.requireValue()?.first
        }
        try ensure(try value?.value.requireValue() == 42, "boundary child mismatch")
        var invalid = [wrap(child + [99]), valid + [99]]
        for end in child.indices { invalid.append(wrap(Array(child.prefix(end)))) }
        for end in valid.indices { invalid.append(Array(valid.prefix(end))) }
        if field == 3 || field == 5 {
          var marker = valid
          marker[columnar ? 3 : 2] = 2
          invalid.append(marker)
        }
        for bytes in invalid {
          var rejected = false
          do { _ = try read(bytes) } catch is LuxoError { rejected = true }
          try ensure(rejected, "accepted malformed nested field \(field), columnar=\(columnar)")
        }
      }
    }
  }

private final class TerminalCounter: @unchecked Sendable {
  private let lock = NSLock()
  private var count = 0
  var value: Int { lock.lock(); defer { lock.unlock() }; return count }
  func increment() { lock.lock(); defer { lock.unlock() }; count += 1 }
}

final class RecordingTransport: Transport, @unchecked Sendable {
  private let base: any Transport
  private let lock = NSLock()
  private var registrations = 0

  init(base: any Transport) { self.base = base }

  var registrationCount: Int {
    lock.lock()
    defer { lock.unlock() }
    return registrations
  }

  func setSchema(_ schema: [String: APISchema]) throws {
    try base.setSchema(schema)
    lock.lock()
    registrations += 1
    lock.unlock()
  }

  func setMode(_ mode: TransportMode) { base.setMode(mode) }
  func setToken(_ token: String) { base.setToken(token) }

  func call(_ api: String, params: [String: LuxoValue]?) async throws -> TransportPayload {
    try await base.call(api, params: params)
  }

  func call(_ api: String, request: any GeneratedRequest) async throws -> TransportPayload {
    try await base.call(api, request: request)
  }

  func subscribe<Value: Sendable>(
    _ api: String, decoder: @escaping @Sendable (TransportPayload) throws -> Value,

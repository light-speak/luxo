func serveHTTP(router *api.Router) error {
	listener, err := net.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		return err
	}
	defer listener.Close()
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	server := &http.Server{
		Handler: api.TraceMiddleware(router), ReadHeaderTimeout: 5 * time.Second,
		BaseContext: func(net.Listener) context.Context { return ctx },
	}
	done := make(chan error, 1)
	go func() { done <- server.Serve(listener) }()
	fmt.Printf("http://%s/luvia\n", listener.Addr())
	// The test owner closes stdin; no fixed port or persistent background daemon.
	_, _ = io.Copy(io.Discard, os.Stdin)
	cancel()
	shutdown, stop := context.WithTimeout(context.Background(), 5*time.Second)
	defer stop()
	if err := server.Shutdown(shutdown); err != nil {
		return err
	}
	if err := <-done; !errors.Is(err, http.ErrServerClosed) {
		return err
	}
	return nil
}

type streamResolver struct{}

func (streamResolver) HandleWatchFiniteJSON(_ context.Context, params *api.StreamParams, _ any, stream *api.TypedStream[[]json.RawMessage]) error {
	value, ok := params.Get("value").(json.RawMessage)
	if !ok {
		return errors.New("missing finite JSON stream value")
	}
	for index := range 64 {
		row := []json.RawMessage{strconv.AppendInt(nil, int64(index), 10), value}
		if err := stream.Send(row); err != nil {
			return err
		}
	}
	if params.Boolean("fail") {
		return NewJSONRejected(value)
	}
	return nil
}

func (streamResolver) HandleWatchJSON(ctx context.Context, params *api.StreamParams, _ any, stream *api.TypedStream[json.RawMessage]) error {
	value, ok := params.Get("value").(json.RawMessage)
	if !ok {
		return errors.New("missing JSON stream value")
	}
	if err := stream.Send(value); err != nil {
		return err
	}
	<-ctx.Done()
	return ctx.Err()
}

func (streamResolver) HandleWatchJSONs(ctx context.Context, params *api.StreamParams, _ any, stream *api.TypedStream[[]json.RawMessage]) error {
	values, ok := params.Get("values").([]json.RawMessage)
	if !ok {
		return errors.New("missing JSON stream values")
	}
	if err := stream.Send(values); err != nil {
		return err
	}
	<-ctx.Done()
	return ctx.Err()
}

func (streamResolver) HandleWatchFixtureState(ctx context.Context, _ *api.StreamParams, _ any, stream *api.TypedStream[FixtureState]) error {
	if err := stream.Send(FixtureState("OPEN")); err != nil {
		return err
	}
	<-ctx.Done()
	return ctx.Err()
}

func (streamResolver) HandleWatchFixtureStates(ctx context.Context, _ *api.StreamParams, _ any, stream *api.TypedStream[[]FixtureState]) error {
	if err := stream.Send([]FixtureState{"OPEN", "CLOSED"}); err != nil {
		return err
	}
	<-ctx.Done()
	return ctx.Err()
}

func (streamResolver) HandleWatchPrecision(ctx context.Context, params *api.StreamParams, _ any, stream *api.TypedStream[Precision]) error {
	value, ok := params.LookupInt("value")
	if !ok {
		return errors.New("missing precision stream value")
	}
	if err := stream.Send(Precision{
		Value: value, Duration: time.Duration(value), Optional: &value,
		Values: []int64{value}, Label: "精度 café", Ignored: 42,
	}); err != nil {
		return err
	}
	<-ctx.Done()
	return ctx.Err()
}

func (streamResolver) HandleWatchNullableArrays(ctx context.Context, params *api.StreamParams, _ any, stream *api.TypedStream[NullableArrays]) error {
	state, ok := params.LookupInt("state")
	if !ok {
		return errors.New("missing nullable array state")
	}
	value := NullableArrays{Tail: 42}
	switch state {
	case 1:
		value.Values, value.Durations, value.Flags = []int64{}, []time.Duration{}, []bool{}
	case 2:
		value.Values = []int64{math.MinInt64, math.MaxInt64}
		value.Durations = []time.Duration{math.MinInt64, math.MaxInt64}
		value.Flags = []bool{true, false}
	}
	if err := stream.Send(value); err != nil {
		return err
	}
	<-ctx.Done()
	return ctx.Err()
}

func (streamResolver) HandleWatchNestedPrecision(ctx context.Context, params *api.StreamParams, _ any, stream *api.TypedStream[NestedPrecision]) error {
	state, ok := params.LookupInt("state")
	if !ok {
		return errors.New("missing nested precision state")
	}
	value := int64(math.MaxInt64)
	row := Precision{
		Value: value, Duration: time.Duration(math.MinInt64), Optional: &value,
		Values: []int64{math.MinInt64, math.MaxInt64}, Label: "精度 café", Ignored: 42,
	}
	group := NestedGroup{Label: "nested 東京", Primary: &row, Rows: []Precision{}, Tail: 42}
	if state > 0 {
		group.Optional = &row
		group.OptionalRows = []Precision{}
	}
	if state == 2 {
		group.Rows, group.OptionalRows = []Precision{row}, []Precision{row}
	}
	if err := stream.Send(NestedPrecision{Group: &group, Groups: []NestedGroup{group}, Tail: -42}); err != nil {
		return err
	}
	<-ctx.Done()
	return ctx.Err()
}

func (streamResolver) HandleWatchScalarArrays(ctx context.Context, params *api.StreamParams, _ any, stream *api.TypedStream[ScalarArrays]) error {
	state, ok := params.LookupInt("state")
	if !ok {
		return errors.New("missing scalar array state")
	}
	value := scalarArrays(state)
	if err := stream.Send(value); err != nil {
		return err
	}
	<-ctx.Done()
	return ctx.Err()
}

func scalarArrays(state int64) ScalarArrays {
	if state == 0 {
		return ScalarArrays{Tail: 42}
	}
	value := ScalarArrays{
		Ints: []int64{math.MinInt64, math.MaxInt64}, Floats: []float64{-1.5, 1.25},
		Strings: []string{"", "精度 café 東京"}, Bools: []bool{true, false},
		Dates:     []time.Time{time.Unix(-1, 0).UTC(), time.Unix(1609459200, 0).UTC()},
		Durations: []time.Duration{math.MinInt64, math.MaxInt64},
		Uuids:     []uuid.UUID{{}, {15: 1}}, Bytes: [][]byte{{}, {0, 127, 128, 255}},
		Decimals: []decimal.Decimal{decimal.New(1234567890123456789, -9), decimal.New(-1, -9)},
		Jsons:    []json.RawMessage{json.RawMessage(`{"value":9223372036854775807,"text":"東京"}`), json.RawMessage(`false`)},
		Enums:    []FixtureState{"OPEN", "CLOSED"}, Tail: 42,
	}
	if state == 1 {
		value.Ints, value.Floats, value.Strings = value.Ints[:0], value.Floats[:0], value.Strings[:0]
		value.Bools, value.Dates, value.Durations = value.Bools[:0], value.Dates[:0], value.Durations[:0]
		value.Uuids, value.Bytes, value.Decimals = value.Uuids[:0], value.Bytes[:0], value.Decimals[:0]
		value.Jsons, value.Enums = value.Jsons[:0], value.Enums[:0]
	}
	if state == 3 {
		value.Floats = []float64{-1.5, 1.25, math.Copysign(0, -1), math.SmallestNonzeroFloat64, -math.SmallestNonzeroFloat64, math.MaxFloat64}
	}
	return value
}
